## 9. CSS変数定義

```css
:root {
  /* 背景色 */
  --bg-primary: #0a0a0a;
  --bg-sidebar: #111111;
  --bg-card: #1a1a1a;
  --bg-hover: #2f2f2f;
  --bg-input: #1f1f1f;
  --bg-modal: #171717;
  --bg-overlay: rgba(0, 0, 0, 0.7);

  /* テキスト色 */
  --text-primary: #ffffff;
  --text-secondary: #a0a0a0;
  --text-muted: #6b7280;
  --text-link: #60a5fa;
  --text-error: #f87171;

  /* ボーダー色 */
  --border-default: #2a2a2a;
  --border-hover: #3f3f3f;
  --border-focus: #3b82f6;
  --border-error: #ef4444;

  /* プライマリカラー */
  --primary: #3b82f6;
  --primary-hover: #2563eb;
  --primary-light: #60a5fa;
  --primary-dark: #1d4ed8;

  /* セカンダリカラー */
  --secondary: #4b5563;
  --secondary-hover: #374151;
  --secondary-light: #6b7280;

  /* ステータスカラー */
  --success: #22c55e;
  --success-bg: #1a2e1a;
  --warning: #eab308;
  --warning-bg: #2e2a1a;
  --error: #ef4444;
  --error-bg: #2e1a1a;
  --info: #3b82f6;
  --info-bg: #1a1a2e;

  /* 温度 */
  --temp-hot: #ef4444;
  --temp-warm: #f59e0b;
  --temp-cold: #3b82f6;

  /* 優先度 */
  --priority-high: #ef4444;
  --priority-medium: #f59e0b;
  --priority-low: #22c55e;

  /* シャドウ */
  --shadow-card: 0 4px 6px rgba(0, 0, 0, 0.3);
  --shadow-modal: 0 10px 25px rgba(0, 0, 0, 0.5);
  --shadow-dropdown: 0 4px 12px rgba(0, 0, 0, 0.4);
}
```
